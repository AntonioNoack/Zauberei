cmake_minimum_required(VERSION 3.16)

project(Zauber LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_package(OpenGL REQUIRED)

add_executable(Zauber
src/CStandardFileIO.c
src/GLSLComputeShader.c
)

target_link_libraries(Zauber PRIVATE OpenGL::GL)

# GLFW
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLFW REQUIRED glfw3)

# GLEW
pkg_check_modules(GLEW REQUIRED glew)

target_include_directories(Zauber PRIVATE
  ${GLFW_INCLUDE_DIRS}
  ${GLEW_INCLUDE_DIRS}
)

target_link_libraries(Zauber PRIVATE
  ${GLFW_LIBRARIES}
  ${GLEW_LIBRARIES}
)
